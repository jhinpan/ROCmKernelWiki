Test Plan

docker build -f ./docker/Dockerfile.rocm_base -t  <IMG_VLLM_ROCM_BASE>  . 

DOCKER_BUILDKIT=1 docker build --no-cache  --target test --progress=plain -t  <FINAL_VLLM_IMAGE_NAME>  -f ./docker/Dockerfile.rocm  --build-
arg BASE_IMAGE=<IMG_VLLM_ROCM_BASE>   .


STEP 2: Please use below command to run VLLM with DeepEP on single node. Select --all2all-backend  as either  deepep_low_latency or deepep_high_throughput.
docker run -it --rm \
  --device /dev/dri \
  --device /dev/kfd \
  --privileged \
  --device /dev/infiniband \
  -v /dev/infiniband:/dev/infiniband \
  --network host \
  --ipc host \
  --group-add video \
  --cap-add SYS_PTRACE \
  --security-opt seccomp=unconfined \
  --ulimit memlock=-1 \
  --shm-size 64G \
  -v <HOST_DS3_MODEL_PATH>:/models/
